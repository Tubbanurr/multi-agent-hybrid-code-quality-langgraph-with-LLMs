# 1️⃣ Branch’ler ve gezinme
git branch                 # Local branch’lerinizi gösterir
git branch <branch_ismi>   # Yeni branch oluşturur (ör. git branch zeynep)
git checkout <branch_ismi> # Belirttiğiniz branch’e geçiş yapar
git checkout -b <branch_ismi> # Yeni branch oluşturup direkt o branch’e geçer

# 2️⃣ Dosya ekleme ve commit
git add <dosya_ismi>       # Belirtilen dosyayı staging area’ya ekler
git add .                  # Tüm değişiklikleri staging area’ya ekler
git commit -m "mesaj"      # Staging area’daki değişiklikleri commit olarak kaydeder
git status                 # Çalışma dizinindeki değişiklikleri gösterir
git log                    # Commit geçmişini gösterir

# 3️⃣ Değişiklikleri göndermek ve almak
git push origin <branch_ismi>  # Local branch’inizi GitHub’a gönderir
git pull origin <branch_ismi>  # GitHub’daki branch’in son değişikliklerini local branch’e çeker
git fetch                       # Remote branch’lerdeki güncellemeleri çeker ama merge etmez
git merge <branch_ismi>         # Belirttiğiniz branch’i bulunduğunuz branch ile birleştirir

# 4️⃣ Dosya silme ve takipten çıkarma
git rm <dosya_ismi>            # Dosyayı siler ve commit’e ekler
git rm --cached <dosya_ismi>   # Dosyayı Git’in takip listesinde çıkarır, local’de kalır

# 5️⃣ Remote branch yönetimi
git remote -v                   # Bağlı olduğunuz remote repository’leri gösterir
git remote add origin <ssh_link> # Local repoyu remote (GitHub) ile bağlar
git push origin --delete <branch_ismi> # GitHub’daki branch’i siler

# 6️⃣ Branch’leri main ile güncelleme (sync)
git checkout main               # Main branch’e geç
git pull origin main            # Güncel main’i çek
git checkout zeynep             # Kendi branch’inize dön
git merge main                  # Main branch’i kendi branch’inize merge et

# 7️⃣ Branch’i main’e ekleme (merge/pull request)
# Terminal ile:
git checkout main
git merge zeynep
git push origin main

# GitHub arayüzü ile (PR):
# 1. zeynep branch’inizi GitHub’a push edin
# 2. GitHub’da repo sayfasına gidin → Compare & pull request → Create pull request → Merge

# 8️⃣ Branch temizliği
git branch -d <branch_ismi>       # Local branch’i siler
git push origin --delete <branch_ismi> # Remote branch’i siler

# 9️⃣ Kullanışlı kısa komutlar
git diff                 # Henüz commit edilmemiş değişiklikleri gösterir
git reset HEAD <dosya>   # Dosyayı staging area’dan çıkarır
git stash                # Çalışmanızı geçici olarak saklar, branch değiştirirken kullanılır
git stash pop            # Saklanan değişikliği geri getirir
